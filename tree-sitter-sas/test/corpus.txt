; Test: KEEP and DROP statements
data work.subset;
  set work.input;
  keep important_var;

; Test: LENGTH statement
data work.typed;
  length name $100 age 8;
  set work.input;
run;

; Test: FORMAT and INFORMAT statements
data work.formatted;
  set work.input;
  format date_var DATE9. amount DOLLAR10.2;
  informat start_date DATE9. salary DOLLAR10.2;
run;

; Test: RETAIN and OUTPUT statements
data work.retained;
  retain row_number 0;
  set work.input;
  row_number + 1;
  if status = 'ACTIVE' then output;
run;

; Test: DELETE statement
data work.filtered;
  set work.input;
  if status = 'INVALID' then delete;
run;
